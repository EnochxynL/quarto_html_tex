% 用 markdown 语法书写列表：统一 itemize 为 leftmargin=*、itemsep=1pt（不显式指定
% topsep）；enumerate 复刻 label=\textbf{P\arabic*.} 的效果。description 暂保留原生 LaTeX。
% 图片 renderer 复刻原生单栏 figure：[!t] 放置、\centering、width=\linewidth、
% caption=title、label={#id} 属性。需 \ExplSyntaxOn 包裹（内部用 \seq_map_inline:Nn）。
\ExplSyntaxOn
\markdownSetup{
  renderers = {
    ulBegin          = {\begin{itemize}[leftmargin=*, itemsep=1pt]},
    ulEnd            = {\end{itemize}},
    olBegin          = {\begin{enumerate}[leftmargin=2em, itemsep=1pt]},
    olItemWithNumber = {\item[\textbf{P#1.}]},
    olEnd            = {\end{enumerate}},
    image            = {
      \begin{figure}[!t]
        \centering
        \includegraphics[width=\linewidth, alt={#1}]{#3}
        \caption{#4}
        \seq_map_inline:Nn \l__markdown_image_identifiers_seq { \label{##1} }
      \end{figure}
    },
  }
}
\ExplSyntaxOff
